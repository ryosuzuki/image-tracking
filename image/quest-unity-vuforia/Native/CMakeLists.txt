cmake_minimum_required(VERSION 3.22.1)
project(quforia LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${VUFORIA_INCLUDE_DIR}/VuforiaEngine/Driver/Driver.h")
  message(FATAL_ERROR "Set VUFORIA_INCLUDE_DIR to privately installed official SDK headers")
endif()
add_library(quforia SHARED src/quforia_jni.cpp src/vuforia_driver.cpp src/external_camera.cpp src/external_tracker.cpp)
target_include_directories(quforia PRIVATE "${VUFORIA_INCLUDE_DIR}" src)
target_link_libraries(quforia android log atomic m c++_shared)
target_compile_options(quforia PRIVATE -Wall -Wextra -Werror=return-type)
set_target_properties(quforia PROPERTIES CXX_VISIBILITY_PRESET default LINK_FLAGS "-Wl,--export-dynamic")
